① 점 무리 짓기 (v6) 🔴 사람이 이해한다

1-1 개념 — 무리 짓기(clustering)

무리 짓기(clustering, 군집화)는 가까이 있는 점끼리 한 무리(cluster)로 묶는 것이다. 방법은 여럿인데, 라이다의 점은 이미 각도 순서로 줄지어 있으므로 가장 간단한 방법이 통한다. 

점을 각도 순서대로 하나씩 본다
   바로 앞 점과의 거리가 임계값보다 가깝다  →  앞 점과 같은 무리
   아니다                                   →  새 무리를 연다

같은 물체의 겉면에 찍힌 점은 이웃끼리 촘촘하고, 다른 물체로 넘어갈 때는 사이가 훌쩍 뛴다는 것에 기댄 방법이다. 그 임계값(파라미터 cluster_gap)을 얼마로 둘지가 문제다.

1-2 숫자를 넣어 본다 — 임계값 10cm

같은 물체 위의 이웃한 두 점 사이는 대략 거리 × 각도 간격(rad)이다. 이 로봇의 간격 1.67도는 0.0291 rad다(23회).

물체까지 거리0.5 m1 m2 m3 m3.4 m
이웃한 두 점 사이1.5 cm2.9 cm5.8 cm8.7 cm9.9 cm
해석: 임계값을 10cm로 두면 3.4m 안쪽의 물체는 한 무리로 묶인다. 그보다 멀면 같은 물체의 점끼리도 10cm 넘게 벌어져 낱낱이 흩어지는데, 어차피 그 거리의 콘은 점이 한두 개라 쓰지 못한다(23회 ⑦절). 반대쪽도 본다. 콘과 콘은 60cm쯤 떨어뜨려 세우므로 겉면 사이가 40cm는 된다. 10cm는 「한 콘 위의 이웃」보다는 넉넉히 크고 「콘과 콘 사이」보다는 넉넉히 작은 값이다. 라이다가 다르거나 콘을 더 촘촘히 세우면 이 숫자를 다시 따져야 한다.

// 27회 cone_drive — v6 (6단계: 스캔을 점으로 펴고, 가까운 점끼리 무리로 묶는다)
#include <algorithm>
#include <chrono>
#include <cmath>
#include <memory>
#include <string>
#include <vector>

#include "rclcpp/rclcpp.hpp"
#include "sensor_msgs/msg/laser_scan.hpp"
#include "geometry_msgs/msg/twist_stamped.hpp"

using namespace std::chrono_literals;

// 평면 위의 점 하나 (m). x 는 앞, y 는 왼쪽 (25회)
struct Point
{
  double x;
  double y;
};

class ConeDrive : public rclcpp::Node
{
public:
  ConeDrive()
  : Node("cone_drive")
  {
    // 믿을 수 있는 거리 범위 — 라이다 사양표에서 가져온다 (24회. LDS-02 기준)
    this->declare_parameter("range_min", 0.16);
    this->declare_parameter("range_max", 8.0);
    range_min_ = this->get_parameter("range_min").as_double();
    range_max_ = this->get_parameter("range_max").as_double();
    // 무리 짓기
    this->declare_parameter("cluster_gap", 0.10);        // 이웃한 두 점이 이보다 멀면 다른 무리다 (m)
    cluster_gap_ = this->get_parameter("cluster_gap").as_double();
    // 라이다는 best effort 로 보낸다 — 받는 쪽도 센서 데이터용 QoS 로 맞춘다 (22회)
    scan_sub_ = this->create_subscription<sensor_msgs::msg::LaserScan>(
      "/scan", rclcpp::SensorDataQoS(),
      [this](sensor_msgs::msg::LaserScan::ConstSharedPtr msg) { on_scan(msg); });
    // 터틀봇3(Jazzy)는 속도 명령을 Twist 가 아니라 TwistStamped 로 받는다
    cmd_pub_ = this->create_publisher<geometry_msgs::msg::TwistStamped>("/cmd_vel", 10);
    timer_ = this->create_wall_timer(100ms, [this]() { on_timer(); });
    RCLCPP_INFO(this->get_logger(), "cluster_gap = %.2f m", cluster_gap_);
  }

private:
  // /scan 이 올 때마다 불린다 (1초에 열 번쯤) — 점 → 무리. 무리마다 점이 몇 개인지를 1초에 한 번 찍는다
  void on_scan(sensor_msgs::msg::LaserScan::ConstSharedPtr msg)
  {
    const std::vector<Point> points = to_points(*msg);
    const std::vector<std::vector<Point>> clusters = make_clusters(points);
    last_scan_time_ = this->now();
    have_scan_ = true;

    std::string text;                            // 무리마다의 점 개수를 각도 순서대로 이어 붙인다
    for (const std::vector<Point> & cluster : clusters) {
      text += " " + std::to_string(cluster.size());
    }
    RCLCPP_INFO_THROTTLE(this->get_logger(), *this->get_clock(), 1000,
                         "점 %zu개 · 무리 %zu개 · 무리마다 점 개수:%s",
                         points.size(), clusters.size(), text.c_str());
  }

  // 타이머가 0.1초마다 부른다 — 아직 판단이 없다. 0 인 명령(「서라」)만 보낸다
  void on_timer()
  {
    geometry_msgs::msg::TwistStamped cmd;        // 속도 여섯 칸이 전부 0 으로 시작한다
    cmd.header.stamp = this->now();
    cmd_pub_->publish(cmd);
    if (!have_scan_ || (this->now() - last_scan_time_).seconds() > scan_timeout_) {
      RCLCPP_WARN_THROTTLE(this->get_logger(), *this->get_clock(), 1000, "스캔이 오지 않습니다");
    }
  }

  // 쓸 수 있는 거리인가 — NaN · inf 가 아니고 사양 범위 안에 있다 (24회)
  bool is_valid(float r) const
  {
    return std::isfinite(r) && r >= range_min_ && r <= range_max_;
  }

  // 스캔의 쓸 수 있는 값을 전부 점으로 편다 — x = r·cos θ, y = r·sin θ (25회)
  std::vector<Point> to_points(const sensor_msgs::msg::LaserScan & scan) const
  {
    std::vector<Point> points;
    points.reserve(scan.ranges.size());
    for (std::size_t i = 0; i < scan.ranges.size(); ++i) {
      const double r = scan.ranges[i];
      if (!is_valid(scan.ranges[i])) {
        continue;
      }
      const double theta = scan.angle_min + i * scan.angle_increment;
      points.push_back(Point{r * std::cos(theta), r * std::sin(theta)});
    }
    return points;
  }

  // 두 점 사이의 거리 (25회)
  double distance(const Point & a, const Point & b) const
  {
    return std::hypot(a.x - b.x, a.y - b.y);
  }

  // 각도 순서로 늘어선 점들을 무리로 묶는다 — 바로 앞 점과 cluster_gap 안이면 같은 무리, 아니면 새 무리
  std::vector<std::vector<Point>> make_clusters(const std::vector<Point> & points) const
  {
    std::vector<std::vector<Point>> clusters;
    for (const Point & p : points) {
      if (clusters.empty() || distance(clusters.back().back(), p) > cluster_gap_) {
        clusters.push_back({});                  // 새 무리를 연다
      }
      clusters.back().push_back(p);
    }
    // 한 바퀴의 끝과 처음은 이어져 있다 — 마지막 무리와 첫 무리가 붙어 있으면 하나로 합친다
    if (clusters.size() > 1 && distance(clusters.back().back(), clusters.front().front()) <= cluster_gap_) {
      clusters.front().insert(clusters.front().begin(), clusters.back().begin(), clusters.back().end());
      clusters.pop_back();
    }
    return clusters;
  }

  double range_min_ = 0.0;        // 파라미터에서 읽는다 (m)
  double range_max_ = 0.0;
  double cluster_gap_ = 0.0;      // 파라미터 cluster_gap (m)
  const double scan_timeout_ = 0.5;         // 스캔이 이 시간(초) 넘게 안 오면 선다
  bool have_scan_ = false;                  // 스캔을 한 번이라도 받았나
  rclcpp::Time last_scan_time_;             // 마지막으로 스캔을 받은 시각

  rclcpp::Subscription<sensor_msgs::msg::LaserScan>::SharedPtr scan_sub_;
  rclcpp::Publisher<geometry_msgs::msg::TwistStamped>::SharedPtr cmd_pub_;
  rclcpp::TimerBase::SharedPtr timer_;
};

int main(int argc, char ** argv)
{
  rclcpp::init(argc, argv);
  auto node = std::make_shared<ConeDrive>();
  rclcpp::spin(node);            // 타이머·구독 콜백은 이 안에서 불린다
  rclcpp::shutdown();
  return 0;
}

// 27회 cone_drive — v7 (7단계: 무리 가운데 콘처럼 생긴 것만 골라 그 중심을 구한다)
#include <algorithm>
#include <chrono>
#include <cmath>
#include <cstdio>
#include <memory>
#include <string>
#include <vector>

#include "rclcpp/rclcpp.hpp"
#include "sensor_msgs/msg/laser_scan.hpp"
#include "geometry_msgs/msg/twist_stamped.hpp"

using namespace std::chrono_literals;

// 평면 위의 점 하나 (m). x 는 앞, y 는 왼쪽 (25회)
struct Point
{
  double x;
  double y;
};

class ConeDrive : public rclcpp::Node
{
public:
  ConeDrive()
  : Node("cone_drive")
  {
    // 믿을 수 있는 거리 범위 — 라이다 사양표에서 가져온다 (24회. LDS-02 기준)
    this->declare_parameter("range_min", 0.16);
    this->declare_parameter("range_max", 8.0);
    range_min_ = this->get_parameter("range_min").as_double();
    range_max_ = this->get_parameter("range_max").as_double();
    // 콘 찾기
    this->declare_parameter("cluster_gap", 0.10);        // 이웃한 두 점이 이보다 멀면 다른 무리다 (m)
    this->declare_parameter("cone_min_points", 3);       // 점이 이보다 적은 무리는 콘으로 보지 않는다
    this->declare_parameter("cone_min_width", 0.05);     // 무리의 폭이 이 사이일 때만 콘이다 (m)
    this->declare_parameter("cone_max_width", 0.40);
    cluster_gap_ = this->get_parameter("cluster_gap").as_double();
    cone_min_points_ = this->get_parameter("cone_min_points").as_int();
    cone_min_width_ = this->get_parameter("cone_min_width").as_double();
    cone_max_width_ = this->get_parameter("cone_max_width").as_double();
    // 라이다는 best effort 로 보낸다 — 받는 쪽도 센서 데이터용 QoS 로 맞춘다 (22회)
    scan_sub_ = this->create_subscription<sensor_msgs::msg::LaserScan>(
      "/scan", rclcpp::SensorDataQoS(),
      [this](sensor_msgs::msg::LaserScan::ConstSharedPtr msg) { on_scan(msg); });
    // 터틀봇3(Jazzy)는 속도 명령을 Twist 가 아니라 TwistStamped 로 받는다
    cmd_pub_ = this->create_publisher<geometry_msgs::msg::TwistStamped>("/cmd_vel", 10);
    timer_ = this->create_wall_timer(100ms, [this]() { on_timer(); });
    RCLCPP_INFO(this->get_logger(), "cluster_gap = %.2f m, 콘: 점 %ld개 이상 · 폭 %.2f ~ %.2f m",
                cluster_gap_, cone_min_points_, cone_min_width_, cone_max_width_);
  }

private:
  // /scan 이 올 때마다 불린다 (1초에 열 번쯤) — 점 → 무리 → 콘. 찾은 콘을 1초에 한 번 찍는다
  void on_scan(sensor_msgs::msg::LaserScan::ConstSharedPtr msg)
  {
    const std::vector<Point> points = to_points(*msg);
    const std::vector<std::vector<Point>> clusters = make_clusters(points);
    const std::vector<Point> cones = find_cones(clusters);
    last_scan_time_ = this->now();
    have_scan_ = true;

    std::string text;                            // 콘의 중심을 가까운 것부터가 아니라 각도 순서대로 이어 붙인다
    for (const Point & c : cones) {
      text += " " + point_text(c);
    }
    RCLCPP_INFO_THROTTLE(this->get_logger(), *this->get_clock(), 1000,
                         "점 %zu개 · 무리 %zu개 · 콘 %zu개:%s",
                         points.size(), clusters.size(), cones.size(), text.c_str());
  }

  // 타이머가 0.1초마다 부른다 — 아직 판단이 없다. 0 인 명령(「서라」)만 보낸다
  void on_timer()
  {
    geometry_msgs::msg::TwistStamped cmd;        // 속도 여섯 칸이 전부 0 으로 시작한다
    cmd.header.stamp = this->now();
    cmd_pub_->publish(cmd);
    if (!have_scan_ || (this->now() - last_scan_time_).seconds() > scan_timeout_) {
      RCLCPP_WARN_THROTTLE(this->get_logger(), *this->get_clock(), 1000, "스캔이 오지 않습니다");
    }
  }

  // 쓸 수 있는 거리인가 — NaN · inf 가 아니고 사양 범위 안에 있다 (24회)
  bool is_valid(float r) const
  {
    return std::isfinite(r) && r >= range_min_ && r <= range_max_;
  }

  // 스캔의 쓸 수 있는 값을 전부 점으로 편다 — x = r·cos θ, y = r·sin θ (25회)
  std::vector<Point> to_points(const sensor_msgs::msg::LaserScan & scan) const
  {
    std::vector<Point> points;
    points.reserve(scan.ranges.size());
    for (std::size_t i = 0; i < scan.ranges.size(); ++i) {
      const double r = scan.ranges[i];
      if (!is_valid(scan.ranges[i])) {
        continue;
      }
      const double theta = scan.angle_min + i * scan.angle_increment;
      points.push_back(Point{r * std::cos(theta), r * std::sin(theta)});
    }
    return points;
  }

  // 두 점 사이의 거리 (25회)
  double distance(const Point & a, const Point & b) const
  {
    return std::hypot(a.x - b.x, a.y - b.y);
  }

  // 각도 순서로 늘어선 점들을 무리로 묶는다 — 바로 앞 점과 cluster_gap 안이면 같은 무리, 아니면 새 무리
  std::vector<std::vector<Point>> make_clusters(const std::vector<Point> & points) const
  {
    std::vector<std::vector<Point>> clusters;
    for (const Point & p : points) {
      if (clusters.empty() || distance(clusters.back().back(), p) > cluster_gap_) {
        clusters.push_back({});                  // 새 무리를 연다
      }
      clusters.back().push_back(p);
    }
    // 한 바퀴의 끝과 처음은 이어져 있다 — 마지막 무리와 첫 무리가 붙어 있으면 하나로 합친다
    if (clusters.size() > 1 && distance(clusters.back().back(), clusters.front().front()) <= cluster_gap_) {
      clusters.front().insert(clusters.front().begin(), clusters.back().begin(), clusters.back().end());
      clusters.pop_back();
    }
    return clusters;
  }

  // 무리 가운데 콘처럼 생긴 것만 골라 그 중심을 돌려준다 — 점이 넉넉하고, 폭이 콘만 하다
  std::vector<Point> find_cones(const std::vector<std::vector<Point>> & clusters) const
  {
    std::vector<Point> cones;
    for (const std::vector<Point> & cluster : clusters) {
      if (static_cast<long>(cluster.size()) < cone_min_points_) {
        continue;                                // 점이 너무 적다 — 잡음일 수 있다
      }
      const double width = distance(cluster.front(), cluster.back());   // 무리의 한쪽 끝에서 다른 쪽 끝까지
      if (width < cone_min_width_ || width > cone_max_width_) {
        continue;                                // 너무 좁거나 넓다 — 벽이나 가구다
      }
      Point sum{0.0, 0.0};
      for (const Point & p : cluster) {
        sum.x += p.x;
        sum.y += p.y;
      }
      cones.push_back(Point{sum.x / cluster.size(), sum.y / cluster.size()});   // 중심 = 점들의 평균
    }
    return cones;
  }

  // 로그에 찍을 글 — 점 하나를 "(x, y)" 로
  std::string point_text(const Point & p) const
  {
    char buf[40];
    std::snprintf(buf, sizeof(buf), "(%.2f, %.2f)", p.x, p.y);
    return buf;
  }

  double range_min_ = 0.0;        // 파라미터에서 읽는다 (m)
  double range_max_ = 0.0;
  double cluster_gap_ = 0.0;      // 파라미터 cluster_gap (m)
  std::int64_t cone_min_points_ = 0;   // 파라미터 cone_min_points
  double cone_min_width_ = 0.0;   // 파라미터 cone_min_width · cone_max_width (m)
  double cone_max_width_ = 0.0;
  const double scan_timeout_ = 0.5;         // 스캔이 이 시간(초) 넘게 안 오면 선다
  bool have_scan_ = false;                  // 스캔을 한 번이라도 받았나
  rclcpp::Time last_scan_time_;             // 마지막으로 스캔을 받은 시각

  rclcpp::Subscription<sensor_msgs::msg::LaserScan>::SharedPtr scan_sub_;
  rclcpp::Publisher<geometry_msgs::msg::TwistStamped>::SharedPtr cmd_pub_;
  rclcpp::TimerBase::SharedPtr timer_;
};

int main(int argc, char ** argv)
{
  rclcpp::init(argc, argv);
  auto node = std::make_shared<ConeDrive>();
  rclcpp::spin(node);            // 타이머·구독 콜백은 이 안에서 불린다
  rclcpp::shutdown();
  return 0;
}

3-1 개념 — 목표점(target point)

콘이 여럿 보일 때 로봇이 향할 한 점을 정한다. 트랙은 왼쪽 콘 줄과 오른쪽 콘 줄 사이이므로, 가장 가까운 왼쪽 콘과 가장 가까운 오른쪽 콘의 가운데를 보고 가면 길 한가운데로 가게 된다.

무엇규칙
볼 콘앞에 있고(x > 0) look_distance(2.0 m) 안에 있는 콘만
좌우 판정콘의 y가 +이면 왼쪽, −이면 오른쪽 (25회 ②절)
짝왼쪽 가운데 가장 가까운 것 하나, 오른쪽 가운데 가장 가까운 것 하나
목표점 — 둘 다 보일 때두 콘의 중점: x끼리, y끼리 더해 2로 나눈다
목표점 — 한쪽만 보일 때보이는 콘에서 트랙 폭의 절반(track_width ÷ 2)만큼 안쪽
하나도 안 보일 때목표점이 없다 — 선다
해석: 다섯째 줄이 빠지면 코너에서 선다. 굽은 길에서는 안쪽 콘이 바깥쪽 콘을 가리거나 한쪽 줄이 시야에서 벗어나 한쪽만 보이는 일이 흔하다. 그때 「짝이 없으니 모르겠다」로 서 버리면 코너를 못 돈다. 왼쪽 콘만 보이면 그 콘의 오른쪽 35cm를, 오른쪽 콘만 보이면 그 왼쪽 35cm를 보고 간다. 이 35cm가 실제 트랙 폭과 맞아야 하므로 파라미터로 빼 둔다.

3-3 8단계 (v8) — 목표점을 정한다

  // 콘들 가운데 조건에 맞는 가장 가까운 것 — want_left 가 true 면 왼쪽(y > 0), false 면 오른쪽(y < 0)
  std::optional<Point> nearest_cone(const std::vector<Point> & cones, bool want_left) const
  {
    std::optional<Point> best;
    for (const Point & c : cones) {
      const bool is_left = c.y > 0.0;
      if (c.x <= 0.0 || is_left != want_left || std::hypot(c.x, c.y) > look_distance_) {
        continue;                                // 뒤에 있거나, 반대쪽이거나, 너무 멀다
      }
      if (!best || std::hypot(c.x, c.y) < std::hypot(best->x, best->y)) {
        best = c;
      }
    }
    return best;
  }

  // 목표점 — 좌우 콘이 다 보이면 그 가운데, 한쪽만 보이면 그 콘에서 트랙 폭의 절반만큼 안쪽
  std::optional<Point> pick_target(const std::vector<Point> & cones)
  {
    const std::optional<Point> left = nearest_cone(cones, true);
    const std::optional<Point> right = nearest_cone(cones, false);
    left_text_ = point_text(left);
    right_text_ = point_text(right);
    std::optional<Point> target;
    if (left && right) {
      target = Point{(left->x + right->x) / 2.0, (left->y + right->y) / 2.0};
    } else if (left) {
      target = Point{left->x, left->y - track_width_ / 2.0};
    } else if (right) {
      target = Point{right->x, right->y + track_width_ / 2.0};
    }
    target_text_ = point_text(target);
    return target;
  }

목표점도 std::optional<Point>다. 콘이 하나도 없으면 비어 있고, 그것이 그대로 「선다」의 조건이 된다. 24회의 「값이 모자라면 숫자를 지어내지 않는다」와 같다. 스캔 콜백은 목표점을 멤버 target_에 적어 두기만 하고, 찍는 것은 타이머가 한다(24회 v5와 같은 짜임).

cone_drive/src/cone_drive.cpp
// 27회 cone_drive — v8 (8단계: 콘을 좌우로 나눠 짝짓고 목표점을 정한다 — 아직 움직이지 않는다)
#include <algorithm>
#include <chrono>
#include <cmath>
#include <cstdio>
#include <memory>
#include <optional>
#include <string>
#include <vector>

#include "rclcpp/rclcpp.hpp"
#include "sensor_msgs/msg/laser_scan.hpp"
#include "geometry_msgs/msg/twist_stamped.hpp"

using namespace std::chrono_literals;

// 평면 위의 점 하나 (m). x 는 앞, y 는 왼쪽 (25회)
struct Point
{
  double x;
  double y;
};

class ConeDrive : public rclcpp::Node
{
public:
  ConeDrive()
  : Node("cone_drive")
  {
    // 믿을 수 있는 거리 범위 — 라이다 사양표에서 가져온다 (24회. LDS-02 기준)
    this->declare_parameter("range_min", 0.16);
    this->declare_parameter("range_max", 8.0);
    range_min_ = this->get_parameter("range_min").as_double();
    range_max_ = this->get_parameter("range_max").as_double();
    // 콘 찾기
    this->declare_parameter("cluster_gap", 0.10);        // 이웃한 두 점이 이보다 멀면 다른 무리다 (m)
    this->declare_parameter("cone_min_points", 3);       // 점이 이보다 적은 무리는 콘으로 보지 않는다
    this->declare_parameter("cone_min_width", 0.05);     // 무리의 폭이 이 사이일 때만 콘이다 (m)
    this->declare_parameter("cone_max_width", 0.40);
    cluster_gap_ = this->get_parameter("cluster_gap").as_double();
    cone_min_points_ = this->get_parameter("cone_min_points").as_int();
    cone_min_width_ = this->get_parameter("cone_min_width").as_double();
    cone_max_width_ = this->get_parameter("cone_max_width").as_double();
    // 목표점
    this->declare_parameter("track_width", 0.70);        // 좌우 콘 사이의 폭 (m) — 한쪽만 보일 때 쓴다
    this->declare_parameter("look_distance", 2.0);       // 이보다 먼 콘은 보지 않는다 (m)
    track_width_ = this->get_parameter("track_width").as_double();
    look_distance_ = this->get_parameter("look_distance").as_double();
    // 라이다는 best effort 로 보낸다 — 받는 쪽도 센서 데이터용 QoS 로 맞춘다 (22회)
    scan_sub_ = this->create_subscription<sensor_msgs::msg::LaserScan>(
      "/scan", rclcpp::SensorDataQoS(),
      [this](sensor_msgs::msg::LaserScan::ConstSharedPtr msg) { on_scan(msg); });
    // 터틀봇3(Jazzy)는 속도 명령을 Twist 가 아니라 TwistStamped 로 받는다
    cmd_pub_ = this->create_publisher<geometry_msgs::msg::TwistStamped>("/cmd_vel", 10);
    timer_ = this->create_wall_timer(100ms, [this]() { on_timer(); });
    RCLCPP_INFO(this->get_logger(), "cluster_gap = %.2f m, 콘: 점 %ld개 이상 · 폭 %.2f ~ %.2f m, track_width = %.2f m, look_distance = %.1f m",
                cluster_gap_, cone_min_points_, cone_min_width_, cone_max_width_, track_width_, look_distance_);
  }

private:
  // /scan 이 올 때마다 불린다 (1초에 열 번쯤) — 점 → 무리 → 콘 → 목표점. 판단은 타이머가 한다
  void on_scan(sensor_msgs::msg::LaserScan::ConstSharedPtr msg)
  {
    const std::vector<Point> points = to_points(*msg);
    const std::vector<std::vector<Point>> clusters = make_clusters(points);
    const std::vector<Point> cones = find_cones(clusters);
    target_ = pick_target(cones);
    last_scan_time_ = this->now();
    have_scan_ = true;
  }

  // 타이머가 0.1초마다 부른다 — 아직 판단이 없다. 0 인 명령(「서라」)만 보내고 목표점을 찍는다
  void on_timer()
  {
    geometry_msgs::msg::TwistStamped cmd;        // 속도 여섯 칸이 전부 0 으로 시작한다
    cmd.header.stamp = this->now();
    cmd_pub_->publish(cmd);

    if (!have_scan_ || (this->now() - last_scan_time_).seconds() > scan_timeout_) {
      RCLCPP_WARN_THROTTLE(this->get_logger(), *this->get_clock(), 1000, "스캔이 오지 않습니다");
      return;
    }
    RCLCPP_INFO_THROTTLE(this->get_logger(), *this->get_clock(), 1000,
                         "왼쪽 콘 %s · 오른쪽 콘 %s · 목표점 %s",
                         left_text_.c_str(), right_text_.c_str(), target_text_.c_str());
  }

  // 쓸 수 있는 거리인가 — NaN · inf 가 아니고 사양 범위 안에 있다 (24회)
  bool is_valid(float r) const
  {
    return std::isfinite(r) && r >= range_min_ && r <= range_max_;
  }

  // 스캔의 쓸 수 있는 값을 전부 점으로 편다 — x = r·cos θ, y = r·sin θ (25회)
  std::vector<Point> to_points(const sensor_msgs::msg::LaserScan & scan) const
  {
    std::vector<Point> points;
    points.reserve(scan.ranges.size());
    for (std::size_t i = 0; i < scan.ranges.size(); ++i) {
      const double r = scan.ranges[i];
      if (!is_valid(scan.ranges[i])) {
        continue;
      }
      const double theta = scan.angle_min + i * scan.angle_increment;
      points.push_back(Point{r * std::cos(theta), r * std::sin(theta)});
    }
    return points;
  }

  // 두 점 사이의 거리 (25회)
  double distance(const Point & a, const Point & b) const
  {
    return std::hypot(a.x - b.x, a.y - b.y);
  }

  // 각도 순서로 늘어선 점들을 무리로 묶는다 — 바로 앞 점과 cluster_gap 안이면 같은 무리, 아니면 새 무리
  std::vector<std::vector<Point>> make_clusters(const std::vector<Point> & points) const
  {
    std::vector<std::vector<Point>> clusters;
    for (const Point & p : points) {
      if (clusters.empty() || distance(clusters.back().back(), p) > cluster_gap_) {
        clusters.push_back({});                  // 새 무리를 연다
      }
      clusters.back().push_back(p);
    }
    // 한 바퀴의 끝과 처음은 이어져 있다 — 마지막 무리와 첫 무리가 붙어 있으면 하나로 합친다
    if (clusters.size() > 1 && distance(clusters.back().back(), clusters.front().front()) <= cluster_gap_) {
      clusters.front().insert(clusters.front().begin(), clusters.back().begin(), clusters.back().end());
      clusters.pop_back();
    }
    return clusters;
  }

  // 무리 가운데 콘처럼 생긴 것만 골라 그 중심을 돌려준다 — 점이 넉넉하고, 폭이 콘만 하다
  std::vector<Point> find_cones(const std::vector<std::vector<Point>> & clusters) const
  {
    std::vector<Point> cones;
    for (const std::vector<Point> & cluster : clusters) {
      if (static_cast<long>(cluster.size()) < cone_min_points_) {
        continue;                                // 점이 너무 적다 — 잡음일 수 있다
      }
      const double width = distance(cluster.front(), cluster.back());   // 무리의 한쪽 끝에서 다른 쪽 끝까지
      if (width < cone_min_width_ || width > cone_max_width_) {
        continue;                                // 너무 좁거나 넓다 — 벽이나 가구다
      }
      Point sum{0.0, 0.0};
      for (const Point & p : cluster) {
        sum.x += p.x;
        sum.y += p.y;
      }
      cones.push_back(Point{sum.x / cluster.size(), sum.y / cluster.size()});   // 중심 = 점들의 평균
    }
    return cones;
  }

  // 콘들 가운데 조건에 맞는 가장 가까운 것 — want_left 가 true 면 왼쪽(y > 0), false 면 오른쪽(y < 0)
  std::optional<Point> nearest_cone(const std::vector<Point> & cones, bool want_left) const
  {
    std::optional<Point> best;
    for (const Point & c : cones) {
      const bool is_left = c.y > 0.0;
      if (c.x <= 0.0 || is_left != want_left || std::hypot(c.x, c.y) > look_distance_) {
        continue;                                // 뒤에 있거나, 반대쪽이거나, 너무 멀다
      }
      if (!best || std::hypot(c.x, c.y) < std::hypot(best->x, best->y)) {
        best = c;
      }
    }
    return best;
  }

  // 목표점 — 좌우 콘이 다 보이면 그 가운데, 한쪽만 보이면 그 콘에서 트랙 폭의 절반만큼 안쪽
  std::optional<Point> pick_target(const std::vector<Point> & cones)
  {
    const std::optional<Point> left = nearest_cone(cones, true);
    const std::optional<Point> right = nearest_cone(cones, false);
    left_text_ = point_text(left);
    right_text_ = point_text(right);
    std::optional<Point> target;
    if (left && right) {
      target = Point{(left->x + right->x) / 2.0, (left->y + right->y) / 2.0};
    } else if (left) {
      target = Point{left->x, left->y - track_width_ / 2.0};
    } else if (right) {
      target = Point{right->x, right->y + track_width_ / 2.0};
    }
    target_text_ = point_text(target);
    return target;
  }

  // 로그에 찍을 글 — 점이 있으면 "(x, y)", 없으면 "없음"
  std::string point_text(const std::optional<Point> & p) const
  {
    if (!p) {
      return "없음";
    }
    char buf[40];
    std::snprintf(buf, sizeof(buf), "(%.2f, %.2f)", p->x, p->y);
    return buf;
  }

  double range_min_ = 0.0;        // 파라미터에서 읽는다 (m)
  double range_max_ = 0.0;
  double cluster_gap_ = 0.0;      // 파라미터 cluster_gap (m)
  std::int64_t cone_min_points_ = 0;   // 파라미터 cone_min_points
  double cone_min_width_ = 0.0;   // 파라미터 cone_min_width · cone_max_width (m)
  double cone_max_width_ = 0.0;
  double track_width_ = 0.0;      // 파라미터 track_width (m)
  double look_distance_ = 0.0;    // 파라미터 look_distance (m)
  const double scan_timeout_ = 0.5;         // 스캔이 이 시간(초) 넘게 안 오면 선다
  bool have_scan_ = false;                  // 스캔을 한 번이라도 받았나
  rclcpp::Time last_scan_time_;             // 마지막으로 스캔을 받은 시각
  std::optional<Point> target_;             // 마지막 스캔에서 정한 목표점 (못 정했으면 비어 있다)
  std::string left_text_ = "없음";         // 로그용 — 마지막 스캔의 왼쪽 콘 · 오른쪽 콘 · 목표점
  std::string right_text_ = "없음";
  std::string target_text_ = "없음";

  rclcpp::Subscription<sensor_msgs::msg::LaserScan>::SharedPtr scan_sub_;
  rclcpp::Publisher<geometry_msgs::msg::TwistStamped>::SharedPtr cmd_pub_;
  rclcpp::TimerBase::SharedPtr timer_;
};

int main(int argc, char ** argv)
{
  rclcpp::init(argc, argv);
  auto node = std::make_shared<ConeDrive>();
  rclcpp::spin(node);            // 타이머·구독 콜백은 이 안에서 불린다
  rclcpp::shutdown();
  return 0;

④ 조향 — 비례 제어와 값 제한 (v9) 🔴 사람이 확인한다

목표점 (x, y)가 정해졌다. 이것을 바퀴 로봇이 받는 두 숫자 — 직진 속도 linear.x와 회전 속도 angular.z — 로 바꾼다.

4-1 개념 — 목표점이 틀어진 각과 비례 제어

먼저 목표점이 내 정면에서 몇 도 틀어져 있는지를 구한다. 25회 「조금 더」의 atan2다. 

틀어진 각 = atan2(목표점의 y, 목표점의 x)        단위 rad.  왼쪽이면 +, 오른쪽이면 −

그리고 가장 간단한 규칙을 쓴다 — 많이 틀어졌으면 많이 돌리고, 조금 틀어졌으면 조금 돌린다. 틀어진 각에 숫자 하나를 곱해 회전 속도로 삼는다. 

회전 속도 = kp × 틀어진 각
직진 속도 = speed (일정)

이렇게 「목표에서 벗어난 양(오차)에 비례해서 고치는 것」을 비례 제어(proportional control, P 제어)라고 하고, 곱하는 숫자 kp를 비례 이득(gain)이라고 한다. 부호는 ROS 좌표 규약이 맞춰 준다 — 목표점이 왼쪽이면 y가 +, 틀어진 각이 +, angular.z가 +, 로봇은 왼쪽으로 돈다. 따로 「왼쪽이면」을 따질 필요가 없다.

4-2 개념 — 값 제한(clamping)

목표점이 순간적으로 엉뚱한 데 잡히면(②절의 가짜 콘) 틀어진 각이 90도가 될 수도 있다. kp가 3이면 회전 속도가 4.7 rad/s — 터틀봇3 버거의 한계(2.84 rad/s)를 넘는 명령이다. 그래서 계산한 값을 정해 둔 범위 안으로 잘라 보낸다. 이것을 값 제한(clamping)이라고 한다. 

      w = std::clamp(kp_ * heading, -max_angular_, max_angular_);       // 비례 제어, 그리고 한계 안으로

std::clamp(값, 아래, 위)는 값이 아래보다 작으면 아래를, 위보다 크면 위를, 그 사이면 값을 그대로 돌려준다(20회 turtle_drive의 angle_to_center에서 한 번 쓰였다). 계산이 튀어도 로봇은 튀지 않는다. 기본 한계 max_angular는 0.5 rad/s로 낮게 잡았다.

4-3 숫자를 넣어 본다

목표점 (0.51, 0.14)        틀어진 각 = atan2(0.14, 0.51) = 0.268 rad = 15.3도
kp = 1.0                   회전 속도 = 1.0 × 0.268 = 0.27 rad/s        한계(0.5) 안 → 그대로
kp = 10                    회전 속도 = 10 × 0.268 = 2.68 rad/s         한계 밖 → 0.50 으로 잘린다

파라미터 셋(speed · kp · max_angular)을 더하고 타이머 콜백을 고친다.

    RCLCPP_INFO_THROTTLE(this->get_logger(), *this->get_clock(), 1000,
                         "왼쪽 콘 %s · 오른쪽 콘 %s · 목표점 %s",

고친 뒤 (v9) — 판단하고, 목표점으로 가는 명령을 보낸다

    double w = 0.0;
    double heading_deg = 0.0;

      why = "콘이 안 보입니다 — 섭니다";

      v = speed_;

    }

    cmd.header.stamp = this->now();
    cmd.twist.linear.x = v;
    cmd.twist.angular.z = w;                     // + 이면 왼쪽으로 돈다
    cmd_pub_->publish(cmd);

// 27회 cone_drive — v9 (9단계: 목표점 쪽으로 조향한다 — 비례 제어 · 값 제한 · 스캔이 끊기면 선다)
#include <algorithm>
#include <chrono>
#include <cmath>
#include <cstdio>
#include <memory>
#include <optional>
#include <string>
#include <vector>

#include "rclcpp/rclcpp.hpp"
#include "sensor_msgs/msg/laser_scan.hpp"
#include "geometry_msgs/msg/twist_stamped.hpp"

using namespace std::chrono_literals;

// 평면 위의 점 하나 (m). x 는 앞, y 는 왼쪽 (25회)
struct Point
{
  double x;
  double y;
};

class ConeDrive : public rclcpp::Node
{
public:
  ConeDrive()
  : Node("cone_drive")
  {
    // 믿을 수 있는 거리 범위 — 라이다 사양표에서 가져온다 (24회. LDS-02 기준)
    this->declare_parameter("range_min", 0.16);
    this->declare_parameter("range_max", 8.0);
    range_min_ = this->get_parameter("range_min").as_double();
    range_max_ = this->get_parameter("range_max").as_double();
    // 콘 찾기
    this->declare_parameter("cluster_gap", 0.10);        // 이웃한 두 점이 이보다 멀면 다른 무리다 (m)
    this->declare_parameter("cone_min_points", 3);       // 점이 이보다 적은 무리는 콘으로 보지 않는다
    this->declare_parameter("cone_min_width", 0.05);     // 무리의 폭이 이 사이일 때만 콘이다 (m)
    this->declare_parameter("cone_max_width", 0.40);
    cluster_gap_ = this->get_parameter("cluster_gap").as_double();
    cone_min_points_ = this->get_parameter("cone_min_points").as_int();
    cone_min_width_ = this->get_parameter("cone_min_width").as_double();
    cone_max_width_ = this->get_parameter("cone_max_width").as_double();
    // 목표점
    this->declare_parameter("track_width", 0.70);        // 좌우 콘 사이의 폭 (m) — 한쪽만 보일 때 쓴다
    this->declare_parameter("look_distance", 2.0);       // 이보다 먼 콘은 보지 않는다 (m)
    track_width_ = this->get_parameter("track_width").as_double();
    look_distance_ = this->get_parameter("look_distance").as_double();
    // 조향
    this->declare_parameter("speed", 0.05);              // 직진 속도 (m/s)
    this->declare_parameter("kp", 1.0);                  // 목표점이 틀어진 각에 곱하는 값
    this->declare_parameter("max_angular", 0.5);         // 회전 속도의 한계 (rad/s)
    speed_ = this->get_parameter("speed").as_double();
    kp_ = this->get_parameter("kp").as_double();
    max_angular_ = this->get_parameter("max_angular").as_double();

    // 라이다는 best effort 로 보낸다 — 받는 쪽도 센서 데이터용 QoS 로 맞춘다 (22회)
    scan_sub_ = this->create_subscription<sensor_msgs::msg::LaserScan>(
      "/scan", rclcpp::SensorDataQoS(),
      [this](sensor_msgs::msg::LaserScan::ConstSharedPtr msg) { on_scan(msg); });
    // 터틀봇3(Jazzy)는 속도 명령을 Twist 가 아니라 TwistStamped 로 받는다
    cmd_pub_ = this->create_publisher<geometry_msgs::msg::TwistStamped>("/cmd_vel", 10);
    timer_ = this->create_wall_timer(100ms, [this]() { on_timer(); });
    RCLCPP_INFO(this->get_logger(), "cluster_gap = %.2f m, 콘: 점 %ld개 이상 · 폭 %.2f ~ %.2f m, track_width = %.2f m, look_distance = %.1f m",
                cluster_gap_, cone_min_points_, cone_min_width_, cone_max_width_, track_width_, look_distance_);
    RCLCPP_INFO(this->get_logger(), "speed = %.2f m/s, kp = %.2f, max_angular = %.2f rad/s", speed_, kp_, max_angular_);
  }

private:
  // /scan 이 올 때마다 불린다 (1초에 열 번쯤) — 점 → 무리 → 콘 → 목표점. 판단은 타이머가 한다
  void on_scan(sensor_msgs::msg::LaserScan::ConstSharedPtr msg)
  {
    const std::vector<Point> points = to_points(*msg);
    const std::vector<std::vector<Point>> clusters = make_clusters(points);
    const std::vector<Point> cones = find_cones(clusters);
    target_ = pick_target(cones);
    last_scan_time_ = this->now();
    have_scan_ = true;
  }

  // 타이머가 0.1초마다 부른다 — 판단하고 명령을 보낸다
  void on_timer()
  {
    double v = 0.0;                              // 보낼 직진 속도와 회전 속도. 「간다」로 정해질 때만 바뀐다
    double w = 0.0;
    double heading_deg = 0.0;
    const char * why = "";
    if (!have_scan_ || (this->now() - last_scan_time_).seconds() > scan_timeout_) {
      why = "스캔이 오지 않습니다 — 섭니다";       // 한 번도 못 받았거나, 마지막으로 받은 지 0.5초가 넘었다
    } else if (!target_) {
      why = "콘이 안 보입니다 — 섭니다";
    } else {
      const double heading = std::atan2(target_->y, target_->x);       // 목표점이 내 앞에서 틀어진 각 (rad). 왼쪽이 +
      heading_deg = heading * 180.0 / M_PI;
      w = std::clamp(kp_ * heading, -max_angular_, max_angular_);       // 비례 제어, 그리고 한계 안으로
      v = speed_;
      why = "목표점으로 갑니다";
    }

    geometry_msgs::msg::TwistStamped cmd;        // 속도 여섯 칸이 전부 0 으로 시작한다
    cmd.header.stamp = this->now();
    cmd.twist.linear.x = v;
    cmd.twist.angular.z = w;                     // + 이면 왼쪽으로 돈다
    cmd_pub_->publish(cmd);

    RCLCPP_INFO_THROTTLE(this->get_logger(), *this->get_clock(), 1000,
                         "왼쪽 콘 %s · 오른쪽 콘 %s · 목표점 %s · 틀어진 각 %.1f도 · 보낸 속도 %.2f m/s, %.2f rad/s · %s",
                         left_text_.c_str(), right_text_.c_str(), target_text_.c_str(), heading_deg, v, w, why);
  }

  // 쓸 수 있는 거리인가 — NaN · inf 가 아니고 사양 범위 안에 있다 (24회)
  bool is_valid(float r) const
  {
    return std::isfinite(r) && r >= range_min_ && r <= range_max_;
  }

  // 스캔의 쓸 수 있는 값을 전부 점으로 편다 — x = r·cos θ, y = r·sin θ (25회)
  std::vector<Point> to_points(const sensor_msgs::msg::LaserScan & scan) const
  {
    std::vector<Point> points;
    points.reserve(scan.ranges.size());
    for (std::size_t i = 0; i < scan.ranges.size(); ++i) {
      const double r = scan.ranges[i];
      if (!is_valid(scan.ranges[i])) {
        continue;
      }
      const double theta = scan.angle_min + i * scan.angle_increment;
      points.push_back(Point{r * std::cos(theta), r * std::sin(theta)});
    }
    return points;
  }

  // 두 점 사이의 거리 (25회)
  double distance(const Point & a, const Point & b) const
  {
    return std::hypot(a.x - b.x, a.y - b.y);
  }

  // 각도 순서로 늘어선 점들을 무리로 묶는다 — 바로 앞 점과 cluster_gap 안이면 같은 무리, 아니면 새 무리
  std::vector<std::vector<Point>> make_clusters(const std::vector<Point> & points) const
  {
    std::vector<std::vector<Point>> clusters;
    for (const Point & p : points) {
      if (clusters.empty() || distance(clusters.back().back(), p) > cluster_gap_) {
        clusters.push_back({});                  // 새 무리를 연다
      }
      clusters.back().push_back(p);
    }
    // 한 바퀴의 끝과 처음은 이어져 있다 — 마지막 무리와 첫 무리가 붙어 있으면 하나로 합친다
    if (clusters.size() > 1 && distance(clusters.back().back(), clusters.front().front()) <= cluster_gap_) {
      clusters.front().insert(clusters.front().begin(), clusters.back().begin(), clusters.back().end());
      clusters.pop_back();
    }
    return clusters;
  }

  // 무리 가운데 콘처럼 생긴 것만 골라 그 중심을 돌려준다 — 점이 넉넉하고, 폭이 콘만 하다
  std::vector<Point> find_cones(const std::vector<std::vector<Point>> & clusters) const
  {
    std::vector<Point> cones;
    for (const std::vector<Point> & cluster : clusters) {
      if (static_cast<long>(cluster.size()) < cone_min_points_) {
        continue;                                // 점이 너무 적다 — 잡음일 수 있다
      }
      const double width = distance(cluster.front(), cluster.back());   // 무리의 한쪽 끝에서 다른 쪽 끝까지
      if (width < cone_min_width_ || width > cone_max_width_) {
        continue;                                // 너무 좁거나 넓다 — 벽이나 가구다
      }
      Point sum{0.0, 0.0};
      for (const Point & p : cluster) {
        sum.x += p.x;
        sum.y += p.y;
      }
      cones.push_back(Point{sum.x / cluster.size(), sum.y / cluster.size()});   // 중심 = 점들의 평균
    }
    return cones;
  }

  // 콘들 가운데 조건에 맞는 가장 가까운 것 — want_left 가 true 면 왼쪽(y > 0), false 면 오른쪽(y < 0)
  std::optional<Point> nearest_cone(const std::vector<Point> & cones, bool want_left) const
  {
    std::optional<Point> best;
    for (const Point & c : cones) {
      const bool is_left = c.y > 0.0;
      if (c.x <= 0.0 || is_left != want_left || std::hypot(c.x, c.y) > look_distance_) {
        continue;                                // 뒤에 있거나, 반대쪽이거나, 너무 멀다
      }
      if (!best || std::hypot(c.x, c.y) < std::hypot(best->x, best->y)) {
        best = c;
      }
    }
    return best;
  }

  // 목표점 — 좌우 콘이 다 보이면 그 가운데, 한쪽만 보이면 그 콘에서 트랙 폭의 절반만큼 안쪽
  std::optional<Point> pick_target(const std::vector<Point> & cones)
  {
    const std::optional<Point> left = nearest_cone(cones, true);
    const std::optional<Point> right = nearest_cone(cones, false);
    left_text_ = point_text(left);
    right_text_ = point_text(right);
    std::optional<Point> target;
    if (left && right) {
      target = Point{(left->x + right->x) / 2.0, (left->y + right->y) / 2.0};
    } else if (left) {
      target = Point{left->x, left->y - track_width_ / 2.0};
    } else if (right) {
      target = Point{right->x, right->y + track_width_ / 2.0};
    }
    target_text_ = point_text(target);
    return target;
  }

  // 로그에 찍을 글 — 점이 있으면 "(x, y)", 없으면 "없음"
  std::string point_text(const std::optional<Point> & p) const
  {
    if (!p) {
      return "없음";
    }
    char buf[40];
    std::snprintf(buf, sizeof(buf), "(%.2f, %.2f)", p->x, p->y);
    return buf;
  }

  double range_min_ = 0.0;        // 파라미터에서 읽는다 (m)
  double range_max_ = 0.0;
  double cluster_gap_ = 0.0;      // 파라미터 cluster_gap (m)
  std::int64_t cone_min_points_ = 0;   // 파라미터 cone_min_points
  double cone_min_width_ = 0.0;   // 파라미터 cone_min_width · cone_max_width (m)
  double cone_max_width_ = 0.0;
  double track_width_ = 0.0;      // 파라미터 track_width (m)
  double look_distance_ = 0.0;    // 파라미터 look_distance (m)
  double speed_ = 0.0;            // 파라미터 speed (m/s)
  double kp_ = 0.0;               // 파라미터 kp
  double max_angular_ = 0.0;      // 파라미터 max_angular (rad/s)
  const double scan_timeout_ = 0.5;         // 스캔이 이 시간(초) 넘게 안 오면 선다
  bool have_scan_ = false;                  // 스캔을 한 번이라도 받았나
  rclcpp::Time last_scan_time_;             // 마지막으로 스캔을 받은 시각
  std::optional<Point> target_;             // 마지막 스캔에서 정한 목표점 (못 정했으면 비어 있다)
  std::string left_text_ = "없음";         // 로그용 — 마지막 스캔의 왼쪽 콘 · 오른쪽 콘 · 목표점
  std::string right_text_ = "없음";
  std::string target_text_ = "없음";

  rclcpp::Subscription<sensor_msgs::msg::LaserScan>::SharedPtr scan_sub_;
  rclcpp::Publisher<geometry_msgs::msg::TwistStamped>::SharedPtr cmd_pub_;
  rclcpp::TimerBase::SharedPtr timer_;
};

int main(int argc, char ** argv)
{
  rclcpp::init(argc, argv);
  auto node = std::make_shared<ConeDrive>();
  rclcpp::spin(node);            // 타이머·구독 콜백은 이 안에서 불린다
  rclcpp::shutdown();
  return 0;
}

4-6 실제로 나온 출력 — 실물 로봇의 바퀴

시뮬레이터에서 맞았다고 실물에서 맞는 것은 아니다. 명령의 부호가 바퀴까지 맞게 가는지를 실물에서 본다. 방법은 이렇다 — 받침 위에 올린 실물 로봇에 v9를 켜되, 스캔은 시뮬레이터가 주는 것을 받게 하고(실물 방에는 콘이 없으므로) 명령은 실물 바퀴로 보낸다. 가상의 로봇을 15도 틀어 세워 목표점이 한쪽에 오게 했다.

목표점이 왼쪽 (0.51, 0.14)틀어진 각 15.3도 · 0.05 m/s, +0.27 rad/s[0.029, 0.070] — 오른쪽 바퀴가 빠르다 → 왼쪽으로 돈다
목표점이 오른쪽 (0.51, −0.13)틀어진 각 −14.7도 · 0.05 m/s, −0.26 rad/s[0.069, 0.028] — 왼쪽 바퀴가 빠르다 → 오른쪽으로 돈다
같은 자세, kp:=10.0틀어진 각 −15.0도 · 0.05 m/s, −0.50 rad/s[0.089, 0.009] — 2.6이 아니라 0.5로 잘렸다
달리는 중에 스캔을 끊는다「스캔이 오지 않습니다 — 섭니다」 · 0.00 m/s끊기 전 [0.051, 0.048] → 2초 뒤 [0.0, 0.0]
해석: 첫 두 줄이 부호다. 왼쪽 목표 → +의 회전 속도 → 오른쪽 바퀴가 더 빨리 돈다. 숫자도 맞는다 — 두 바퀴 사이가 16cm이므로 바퀴의 빠르기는 0.05 ∓ 0.27 × 0.08 = 0.028과 0.072다. 셋째 줄이 값 제한, 넷째 줄이 타임아웃 정지다. 넷 다 바퀴를 띄운 실물 로봇에서 /joint_states로 읽은 값이다. 이 확인을 건너뛰고 바닥에 내려놓으면, 부호가 뒤집힌 로봇은 콘에서 멀어지는 쪽이 아니라 콘 쪽으로 돈다.

⑤ kp 튜닝 🔴 사람이 맞춘다

kp는 「틀어진 것을 얼마나 세게 고칠 것인가」다. 작으면 굼뜨고 크면 과하다. 어디가 알맞은지는 식으로 나오지 않고 돌려 봐야 안다. 시뮬레이터의 같은 트랙에서 kp만 바꿔 네 번 달렸다. 차이가 드러나도록 회전 속도의 한계는 2.0 rad/s로 풀었다.

3.090.7초0개2.00 rad/s (한계)0.103 rad/s
8.090.6초0개2.00 rad/s (한계)0.262 rad/s
해석: 뜻밖에 네 값 모두 콘에 닿지 않고 같은 시간에 끝까지 갔다. 이 트랙이 완만하고, 시뮬레이터의 로봇은 명령을 받는 즉시 그대로 움직이기 때문이다. 차이는 명령의 출렁임에서 났다. 콘 짝을 지나칠 때마다 목표점이 한 칸 앞으로 건너뛰고 틀어진 각이 갑자기 달라지는데, kp 0.3은 그것을 0.1 rad/s 남짓의 완만한 명령으로 받아넘기고 kp 8은 한계값 ±2.0 rad/s를 몇 번씩 오가는 명령으로 바꾼다. 명령이 한 번 바뀔 때의 크기가 40배쯤 다르다. 실물 로봇은 바퀴가 그렇게 빨리 따라가지 못하므로 이런 명령을 받으면 몸체가 좌우로 떨 것이라고 짐작할 수 있다. 다만 그것은 짐작이다 — 어느 값이 알맞은지는 이 시뮬레이터로는 가려지지 않았고, 실물 트랙에서 직접 봐야 한다.

그래서 이 값들은 처음부터 파라미터로 빼 두었다(21회). 다시 빌드하지 않고 켤 때 바꿔 가며 맞춘다. 

$ ros2 run cone_drive cone_drive --ros-args -p kp:=3.0 -p max_angular:=1.0
⚠️ 시뮬레이터에서 맞춘 값이 실물에서도 맞는 것은 아니다

시뮬레이터의 로봇은 명령을 받는 즉시 그 속도가 되고, 바퀴가 미끄러지지 않고, 콘은 정확히 그 자리에 있다. 실물은 바퀴가 속도를 따라가는 데 시간이 걸리고, 바닥에서 미끄러지고, 콘은 밀리고 기운다. 시뮬레이터는 논리를 맞추는 곳이지 숫자를 맞추는 곳이 아니다. 부호가 맞는가, 한쪽 콘만 보일 때도 가는가, 스캔이 끊기면 서는가 — 여기까지를 시뮬레이터에서 보고, kp와 속도는 실물 트랙 앞에서 다시 맞춘다. 그 일은 에이전트가 대신해 줄 수 없다.

cone_track_sim.py
#!/usr/bin/env python3
"""27회 — 콘 트랙을 흉내 내는 작은 시뮬레이터.

콘의 자리를 정해 두고, 가상의 로봇이 그 사이에 있을 때 LDS-02 가 내놓을 /scan 을 계산해서 발행한다.
/cmd_vel(TwistStamped)을 받아 가상의 로봇을 움직이므로, 주행 노드를 붙이면 닫힌 고리로 돌아간다.
실물 라이다를 닮게 했다 — 한 바퀴 216개쯤(프레임마다 다르다), angle_min 이 조금씩 다르고,
아무것도 없는 방향은 NaN, range_min/range_max 는 0.0/100.0, 거리에는 작은 잡음.

쓰는 법 (실물 로봇의 /scan 과 섞이지 않게 다른 도메인에서):
    export ROS_DOMAIN_ID=77
    python3 cone_track_sim.py                         # 기본 트랙
    python3 cone_track_sim.py --ros-args -p track:=straight -p scan_topic:=/scan
파라미터: track(curve·straight·left_only), scan_topic, x0·y0·yaw0_deg(로봇의 처음 자세), noise(m), log_csv(경로)
"""
import math, random

import rclpy
from rclpy.node import Node
from rclpy.qos import qos_profile_sensor_data
from sensor_msgs.msg import LaserScan
from geometry_msgs.msg import TwistStamped

CONE_RADIUS = 0.10          # 지름 20 cm
RANGE_MIN, RANGE_MAX = 0.16, 8.0      # LDS-02 사양 — 이 밖은 NaN 으로 낸다
SAMPLE_RATE, SCAN_HZ = 2300.0, 10.6   # 23회에서 본 값
HALF_WIDTH = 0.35           # 트랙 폭 70 cm 의 절반


def make_track(kind):
    """콘의 자리 [(x, y, '왼'/'오')…] 와 길의 가운데 선 [(x, y)…]"""
    center = []
    if kind == 'straight':
        center = [(0.3 * i, 0.0, 0.0) for i in range(0, 14)]
    else:  # curve — 1.2 m 직진 뒤 반지름 1.5 m 로 왼쪽으로 90도, 다시 1.2 m 직진
        s = 0.0
        while s <= 1.2: center.append((s, 0.0, 0.0)); s += 0.1
        R, n = 1.5, 24
        for k in range(1, n + 1):
            a = (math.pi / 2) * k / n
            center.append((1.2 + R * math.sin(a), R - R * math.cos(a), a))
        for k in range(1, 13):
            center.append((1.2 + R, R + 0.1 * k, math.pi / 2))
    cones, last = [], None
    for (x, y, th) in center:                     # 가운데 선을 따라 60 cm 쯤마다 좌우에 하나씩
        if last is not None and math.hypot(x - last[0], y - last[1]) < 0.6: continue
        last = (x, y)
        if x < 0.25 and y < 0.25: continue        # 출발 자리 바로 옆은 비운다
        nx, ny = -math.sin(th), math.cos(th)      # 진행 방향의 왼쪽
        if kind != 'right_only': cones.append((x + HALF_WIDTH * nx, y + HALF_WIDTH * ny, '왼'))
        if kind != 'left_only': cones.append((x - HALF_WIDTH * nx, y - HALF_WIDTH * ny, '오'))
    return cones, [(x, y) for x, y, _ in center]


class ConeTrackSim(Node):
    def __init__(self):
        super().__init__('cone_track_sim')
        self.declare_parameter('track', 'curve'); self.declare_parameter('scan_topic', '/scan')
        self.declare_parameter('x0', 0.0); self.declare_parameter('y0', 0.0); self.declare_parameter('yaw0_deg', 0.0)
        self.declare_parameter('noise', 0.003); self.declare_parameter('log_csv', ''); self.declare_parameter('frozen', False)
        g = lambda n: self.get_parameter(n).value
        self.cones, self.center = make_track(g('track'))
        self.x, self.y, self.yaw = g('x0'), g('y0'), math.radians(g('yaw0_deg'))
        self.noise, self.frozen = g('noise'), g('frozen')
        self.v = self.w = 0.0; self.last_cmd = None; self.t = 0.0; self.touched = set(); self.done = False
        self.csv = open(g('log_csv'), 'w') if g('log_csv') else None
        if self.csv: self.csv.write('t,x,y,yaw,v,w,min_cone_dist\n')
        self.pub = self.create_publisher(LaserScan, g('scan_topic'), qos_profile_sensor_data)
        self.create_subscription(TwistStamped, '/cmd_vel', self.on_cmd, 10)
        self.dt = 0.02
        self.create_timer(self.dt, self.step); self.create_timer(1.0 / SCAN_HZ, self.publish_scan); self.create_timer(1.0, self.report)
        self.get_logger().info(f"트랙 {g('track')} · 콘 {len(self.cones)}개 · 로봇 처음 자세 ({self.x:.2f}, {self.y:.2f}, {g('yaw0_deg'):.0f}도)")

    def on_cmd(self, msg):
        self.v, self.w = msg.twist.linear.x, msg.twist.angular.z
        self.last_cmd = self.get_clock().now()

    def step(self):          # 차동 구동 로봇의 운동 — 실물처럼 마지막 명령을 계속 따른다
        self.t += self.dt
        if not self.frozen:
            self.x += self.v * math.cos(self.yaw) * self.dt
            self.y += self.v * math.sin(self.yaw) * self.dt
            self.yaw += self.w * self.dt
        d = min(math.hypot(cx - self.x, cy - self.y) for cx, cy, _ in self.cones)
        for i, (cx, cy, _) in enumerate(self.cones):
            if math.hypot(cx - self.x, cy - self.y) < CONE_RADIUS + 0.09 and i not in self.touched:   # 버거의 반지름 9 cm 쯤
                self.touched.add(i); self.get_logger().warn(f'콘에 닿았습니다 — ({cx:.2f}, {cy:.2f})')
        if self.csv: self.csv.write(f'{self.t:.2f},{self.x:.4f},{self.y:.4f},{self.yaw:.4f},{self.v:.3f},{self.w:.3f},{d:.4f}\n')
        ex, ey = self.center[-1]
        if not self.done and math.hypot(ex - self.x, ey - self.y) < 0.3:
            self.done = True; self.get_logger().info(f'트랙 끝에 닿았습니다 — {self.t:.1f}초, 닿은 콘 {len(self.touched)}개')

    def report(self):
        self.get_logger().info(f'로봇 ({self.x:.2f}, {self.y:.2f}, {math.degrees(self.yaw):.0f}도) · 받은 속도 {self.v:.2f} m/s, {self.w:.2f} rad/s')

    def publish_scan(self):
        speed = 360.0 * SCAN_HZ * random.uniform(0.995, 1.005)          # 도/초
        inc = math.radians(speed / SAMPLE_RATE)
        a0 = random.uniform(0.0, inc)
        n = int(math.ceil((2 * math.pi - a0 - 0.02) / inc))
        ranges = [float('nan')] * n
        for i in range(n):
            th = self.yaw + a0 + i * inc
            dx, dy = math.cos(th), math.sin(th); best = None
            for cx, cy, _ in self.cones:               # 광선과 원의 교점
                ox, oy = cx - self.x, cy - self.y
                b = ox * dx + oy * dy
                disc = b * b - (ox * ox + oy * oy - CONE_RADIUS ** 2)
                if b <= 0 or disc < 0: continue
                r = b - math.sqrt(disc)
                if best is None or r < best: best = r
            if best is not None and RANGE_MIN <= best <= RANGE_MAX:
                ranges[i] = round(best * random.gauss(1.0, 0.002) + random.gauss(0.0, self.noise), 3)
        m = LaserScan()
        m.header.stamp = self.get_clock().now().to_msg(); m.header.frame_id = 'base_scan'
        m.angle_min, m.angle_increment, m.angle_max = a0, inc, a0 + (n - 1) * inc
        m.scan_time = 1.0 / SCAN_HZ; m.time_increment = m.scan_time / n
        m.range_min, m.range_max = 0.0, 100.0
        m.ranges = ranges; m.intensities = [float('nan') if math.isnan(r) else 220.0 for r in ranges]
        self.pub.publish(m)


def main():
    rclpy.init(); node = ConeTrackSim()
    try: rclpy.spin(node)
    except KeyboardInterrupt: pass
    finally:
        if node.csv: node.csv.close()


if __name__ == '__main__':
    main()

콘의 중심 좌표 목록(로봇 기준, x 앞, y 왼쪽)을 받아서 목표점 하나를 정하는 함수를 만들어 줘. 앞에 있고 look_distance 안인 콘만 본다. 왼쪽(y > 0)에서 가장 가까운 것과 오른쪽(y < 0)에서 가장 가까운 것의 중점이 목표점이다. 한쪽만 있으면 그 콘에서 track_width의 절반만큼 안쪽이다. 하나도 없으면 std::nullopt를 돌려줘. 0이나 -1 같은 값으로 「없음」을 나타내지 마.

✅ 오늘의 점검

조금 더 알아두면 시험 범위 아님

P 다음의 I와 D — PID 제어

비례 제어는 「지금 벗어난 양」만 본다. 여기에 벗어난 양이 쌓인 것(적분, I)과 벗어난 양이 바뀌는 빠르기(미분, D)를 더해 쓰는 것이 PID 제어다. I는 조금씩 계속 남는 오차를 없애고, D는 목표에 가까워질 때 미리 힘을 빼서 출렁임을 줄인다. 이 프로젝트의 속도에서는 P만으로 대체로 충분하고, kp를 올렸을 때 출렁임이 문제가 되면 D를 생각해 볼 수 있다. 제어공학 과목에서 자세히 다룬다.

굽은 길에서 속도 줄이기

v9는 직진 속도가 일정하다. 사람은 굽은 길에서 속도를 줄인다. 같은 일을 한 줄로 할 수 있다 — 틀어진 각이 클수록 직진 속도를 낮춘다. 예를 들어 v = speed_ * std::max(0.3, std::cos(heading))로 두면 정면(0도)에서는 그대로, 60도 틀어졌을 때 절반, 그 뒤로는 30%까지만 줄어든다. 프로젝트에서 완주 시간을 줄이려 할 때 손댈 만한 자리다. 이 줄은 설명용이고 오늘의 단계 코드에는 넣지 않았다.

TF2는 어디 갔나

26회에서 말한 대로 v9는 점을 base_link로 옮기지 않고 라이다 기준(base_scan) 그대로 쓴다. 3.2cm 차이가 목표점의 각에 주는 영향은 50cm 앞의 목표점에서 1도 안팎이다. 옮기고 싶다면 to_points가 돌려준 점들을 26회의 방법으로 옮기면 되고, 점이 많으므로 lookupTransform으로 관계를 한 번 받아 두고 쓰는 쪽이 낫다. 센서가 둘이 되거나 라이다가 몸체에서 멀어지면 그때는 꼭 필요하다.
